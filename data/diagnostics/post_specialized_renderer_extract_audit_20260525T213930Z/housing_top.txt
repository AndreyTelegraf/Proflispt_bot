   104	def _hs_choice_kb(step, slug: str) -> InlineKeyboardMarkup:
   105	    b = InlineKeyboardBuilder()
   106	    field = getattr(step, "field_name", "")
   107	    label_defaults = {"yes": "Да", "no": "Нет"}
   108	    for opt in getattr(step, "options", []) or []:
   109	        val = str(opt.get("value", "")).strip()
   110	        if not val:
   111	            continue
   112	        label = str(opt.get("label") or label_defaults.get(val.lower(), val)).strip()
   113	        b.add(InlineKeyboardButton(text=label, callback_data=f"hs:choice:{slug}:{val}"))
   114	    is_residency = (field == "resides_in_portugal")
   115	    if not is_residency:
   116	        b.add(InlineKeyboardButton(text="← Назад", callback_data=f"hs:back:{slug}"))
   117	    if field == "geo_tags":
   118	        b.adjust(3, 3, 3, 1, 1)
   119	    elif is_residency:
   120	        b.adjust(2)
