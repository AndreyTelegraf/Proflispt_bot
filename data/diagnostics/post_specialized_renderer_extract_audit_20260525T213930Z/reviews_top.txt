    69	def _normalize_geo(value: str) -> str:
    70	    raw = str(value or "").strip()
    71	    if not raw:
    72	        return json.dumps([], ensure_ascii=False)
    73	    parts = [c.strip().lstrip("#").lower() for c in raw.replace(",", " ").split() if c.strip()]
    74	    return json.dumps(parts, ensure_ascii=False)
