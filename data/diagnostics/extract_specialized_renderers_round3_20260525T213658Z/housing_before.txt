    91	def _hs_render_html(payload: dict) -> str:
    92	    """Housing-specific render: appends rental_term tag to geo line, omits review links."""
    93	    lines: list[str] = []
    94	
    95	    geo_line_raw = render_geo_tags(payload.get("geo_tags"))
    96	    rental_tag = _RENTAL_TAGS.get(str(payload.get("rental_term", "")), "")
    97	
    98	    if geo_line_raw:
    99	        geo_line = html.escape(geo_line_raw)
   100	        if rental_tag:
   101	            geo_line += " " + html.escape(rental_tag)
   102	        lines.append(geo_line)
   103	
   104	    description = _norm(payload.get("description"))
   105	    if description:
   106	        dlines = [p.strip() for p in description.splitlines() if p.strip()]
   107	        if dlines:
   108	            lines.append("- " + html.escape(dlines[0]))
   109	            for p in dlines[1:]:
   110	                lines.append(html.escape(p))
   111	
   112	    for link in _split_lines(payload.get("social_links")):
   113	        lines.append(f'<a href="{html.escape(link, quote=True)}">Ссылка</a>')
   114	
   115	    tg = _norm(payload.get("telegram"))
   116	    if tg:
   117	        lines.append(html.escape(tg))
   118	
   119	    pm = _norm(payload.get("phone_main"))
   120	    if pm:
   121	        lines.append(html.escape(pm))
   122	
   123	    pw = _norm(payload.get("phone_whatsapp"))
   124	    if pw and pw != pm:
   125	        lines.append(html.escape(pw))
   126	
   127	    cn = _norm(payload.get("contact_name"))
   128	    if cn:
   129	        lines.append("- " + html.escape(cn))
   130	
   131	    return "\n".join(lines).strip()
