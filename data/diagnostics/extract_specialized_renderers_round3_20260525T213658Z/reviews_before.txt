    69	def _rv_render_html(payload: dict) -> str:
    70	    lines: list[str] = []
    71	
    72	    geo = str(payload.get("geo_tags") or "").strip()
    73	    if geo:
    74	        tag = geo if geo.startswith("#") else f"#{geo.lstrip('#').lower()}"
    75	        lines.append(html.escape(tag))
    76	
    77	    performer = str(payload.get("performer_contacts") or "").strip()
    78	    if performer:
    79	        uname = performer if performer.startswith("@") else f"@{performer.lstrip('@')}"
    80	        lines.append(f"Исполнитель: {html.escape(uname)}")
    81	
    82	    description = str(payload.get("description") or "").strip()
    83	    if description:
    84	        dlines = [p.strip() for p in description.splitlines() if p.strip()]
    85	        if dlines:
    86	            lines.append("- " + html.escape(dlines[0]))
    87	            for p in dlines[1:]:
    88	                lines.append(html.escape(p))
    89	
    90	    author = str(payload.get("author_telegram") or "").strip()
    91	    if author:
    92	        uname = author if author.startswith("@") else f"@{author.lstrip('@')}"
    93	        lines.append(f"Автор: {html.escape(uname)}")
    94	
    95	    return "\n".join(lines).strip()

    98	def _rv_render_html_from_post(post: dict) -> str:
    99	    cities_raw = post.get("cities")
   100	    geo_tags = ""
   101	    if cities_raw:
   102	        try:
   103	            cities = json.loads(cities_raw) if isinstance(cities_raw, str) else cities_raw
   104	            if isinstance(cities, list):
   105	                geo_tags = " ".join(
   106	                    f"#{str(x).strip().lstrip('#').lower()}"
   107	                    for x in cities if str(x).strip()
   108	                )
   109	            else:
   110	                geo_tags = str(cities_raw).strip()
   111	        except Exception:
   112	            geo_tags = str(cities_raw).strip()
   113	
   114	    return _rv_render_html({
   115	        "geo_tags": geo_tags,
   116	        "description": post.get("description", ""),
   117	        "performer_contacts": post.get("social_media", ""),
   118	        "author_telegram": post.get("telegram_username", ""),
   119	    })

   122	def _normalize_geo(value: str) -> str:
   123	    raw = str(value or "").strip()
   124	    if not raw:
   125	        return json.dumps([], ensure_ascii=False)
   126	    parts = [c.strip().lstrip("#").lower() for c in raw.replace(",", " ").split() if c.strip()]
   127	    return json.dumps(parts, ensure_ascii=False)
